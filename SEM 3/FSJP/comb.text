class ShapeArea {
    double calculateArea(double radius) {
        return Math.PI * radius * radius;
    }

    double calculateArea(double length, double breadth) {
        return length * breadth;
    }

    double calculateArea(float base, float height) {
        return 0.5 * base * height;
    }

    double calculateArea(double a, double b, double c) {
        double s = (a + b + c) / 2.0;
        return Math.sqrt(s * (s - a) * (s - b) * (s - c));
    }
}

A bank account should not allow withdrawal amounts greater than the available balance. Develop a program that handles such invalid transactions using exception handling.

class InsufficientFundsException extends Exception {
    private double requestedAmount;
    private double currentBalance;

    public InsufficientFundsException(String message, double requestedAmount, double currentBalance) {
        super(message);
        this.requestedAmount = requestedAmount;
        this.currentBalance = currentBalance;
    }

    public double getDeficit() {
        return requestedAmount - currentBalance;
    }

    public double getRequestedAmount() {
        return requestedAmount;
    }

    public double getCurrentBalance() {
        return currentBalance;
    }
}

class BankAccount {
    private String accountNumber;
    private String accountHolder;
    private double balance;

    public BankAccount(String accountNumber, String accountHolder, double initialBalance) {
        this.accountNumber = accountNumber;
        this.accountHolder = accountHolder;
        this.balance = Math.max(0.0, initialBalance);
    }

    public void deposit(double amount) {
        if (amount <= 0) {
            System.out.println("Invalid deposit amount! Amount must be positive.");
            return;
        }
        balance += amount;
        System.out.printf("Deposited: Rs. %.2f | Updated Balance: Rs. %.2f%n", amount, balance);
    }

    public void withdraw(double amount) throws InsufficientFundsException {
        System.out.printf("Attempting to withdraw: Rs. %.2f...%n", amount);

        if (amount <= 0) {
            System.out.println("Invalid transaction: Withdrawal amount must be greater than 0.");
            return;
        }

        if (amount > balance) {
            throw new InsufficientFundsException(
                "Transaction Declined: Insufficient balance in account.",
                amount,
                balance
            );
        }

        balance -= amount;
        System.out.printf("Withdrawal Successful: Rs. %.2f deducted. Remaining Balance: Rs. %.2f%n", amount, balance);
    }

    public void displaySummary() {
        System.out.println("----------------------------------------");
        System.out.println("Account Number : " + accountNumber);
        System.out.println("Account Holder : " + accountHolder);
        System.out.printf("Current Balance: Rs. %.2f%n", balance);
        System.out.println("----------------------------------------");
    }
}

            System.out.printf("  Available Balance: Rs. %.2f%n", e.getCurrentBalance());

A railway reservation system performs ticket booking and ticket cancellation simultaneously. Implement a program using multiple threads to handle both tasks.

class RailwaySystem {
    private int availableTickets = 10;

    public synchronized void bookTicket(String passengerName, int seats) {
        if (seats <= 0) {
            System.out.println("Invalid booking request by " + passengerName);
            return;
        }
        if (availableTickets >= seats) {
            System.out.println(passengerName + " is booking " + seats + " ticket(s)...");
            try {
                Thread.sleep(300);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
            availableTickets -= seats;
            System.out.println("Booking Confirmed for " + passengerName + "! Remaining tickets: " + availableTickets);
        } else {
            System.out.println("Booking Failed for " + passengerName + "! Requested: " + seats + ", Available: " + availableTickets);
        }
        System.out.println("----------------------------------------");
    }

    public synchronized void cancelTicket(String passengerName, int seats) {
        if (seats <= 0) {
            System.out.println("Invalid cancellation request by " + passengerName);
            return;
        }
        System.out.println(passengerName + " is cancelling " + seats + " ticket(s)...");
        try {
            Thread.sleep(200);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }
        availableTickets += seats;
        System.out.println("Cancellation Confirmed for " + passengerName + "! Remaining tickets: " + availableTickets);
        System.out.println("----------------------------------------");
    }

    public int getAvailableTickets() {
        return availableTickets;
    }
}

class BookingThread extends Thread {
    private RailwaySystem system;
    private String passengerName;
    private int seats;

    public BookingThread(RailwaySystem system, String passengerName, int seats) {
        this.system = system;
        this.passengerName = passengerName;
        this.seats = seats;
    }

    @Override
    public void run() {
        system.bookTicket(passengerName, seats);
    }
}

class CancellationThread extends Thread {
    private RailwaySystem system;
    private String passengerName;
    private int seats;

    public CancellationThread(RailwaySystem system, String passengerName, int seats) {
        this.system = system;
        this.passengerName = passengerName;
        this.seats = seats;
    }

    @Override
    public void run() {
        system.cancelTicket(passengerName, seats);
    }
}

class MultithreadingDemo {
    public static void main(String[] args) {
        System.out.println("--- Railway Reservation System (Multithreading) ---\n");

        RailwaySystem railway = new RailwaySystem();
        System.out.println("Initial Available Tickets: " + railway.getAvailableTickets());
        System.out.println("========================================");

        BookingThread b1 = new BookingThread(railway, "Alice", 4);
        CancellationThread c1 = new CancellationThread(railway, "Bob", 2);
        BookingThread b2 = new BookingThread(railway, "Charlie", 6);
        CancellationThread c2 = new CancellationThread(railway, "David", 1);
        BookingThread b3 = new BookingThread(railway, "Eva", 3);

        b1.start();
        c1.start();
        b2.start();
        c2.start();
        b3.start();

        try {
            b1.join();
            c1.join();
            b2.join();
            c2.join();
            b3.join();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        System.out.println("All transactions finished.");
        System.out.println("Final Available Tickets: " + railway.getAvailableTickets());
    }
}

import java.util.regex.Matcher;
import java.util.regex.Pattern;

class NewsletterService {
    private static final String EMAIL_REGEX = "^[a-zA-Z0-9_+&*-]+(?:\\.[a-zA-Z0-9_+&*-]+)*@(?:[a-zA-Z0-9-]+\\.)+[a-zA-Z]{2,7}$";
    private static final Pattern pattern = Pattern.compile(EMAIL_REGEX);

    public boolean validateEmail(String email) {
        if (email == null || email.trim().isEmpty()) {
            return false;
        }
        Matcher matcher = pattern.matcher(email.trim());
        return matcher.matches();
    }

    public void processSubscription(String userEmail) {
        System.out.println("Processing subscription request for: " + userEmail);
        if (validateEmail(userEmail)) {
            System.out.println("Status: Valid Email Address.");
            System.out.println("Result: Subscription accepted! Welcome to the Newsletter.");
        } else {
            System.out.println("Status: Invalid Email Format.");
            System.out.println("Result: Subscription rejected! Please enter a valid email address.");
        }
        System.out.println("----------------------------------------");
    }
}

A website designer wants the webpage background color to change automatically every five seconds to improve visual appeal. Write a JavaScript program using DOM to implement this functionality.

<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Automatic Background Color Changer</title>
    <style>
        body {
            font-family: Arial, sans-serif;
            display: flex;
            justify-content: center;
            align-items: center;
            height: 100vh;
            margin: 0;
            transition: background-color 1s ease;
            background-color: #f0f4f8;
        }
        .box {
            background: rgba(255, 255, 255, 0.95);
            padding: 30px;
            border-radius: 14px;
            box-shadow: 0 8px 24px rgba(0,0,0,0.15);
            text-align: center;
            max-width: 400px;
            width: 100%;
        }
        h2 {
            color: #333;
            margin-bottom: 12px;
        }
        #currentColor {
            font-size: 1.2rem;
            font-weight: bold;
            color: #2b6cb0;
            margin: 15px 0;
        }
        #timerNotice {
            font-size: 0.9rem;
            color: #666;
        }
    </style>
</head>
<body>
    <div class="box">
        <h2>Dynamic Color Changer</h2>
        <p id="timerNotice">Changes automatically every 5 seconds</p>
        <div id="currentColor">Current Color: Light Blue (#dbeafe)</div>
    </div>

    <script>
        const colors = [
            { name: "Light Blue", hex: "#dbeafe" },
            { name: "Mint Green", hex: "#d1fae5" },
            { name: "Pale Peach", hex: "#ffedd5" },
            { name: "Lavender", hex: "#ede9fe" },
            { name: "Light Rose", hex: "#ffe4e6" },
            { name: "Sunny Yellow", hex: "#fef9c3" }
        ];

        let currentIndex = 0;

        function changeBackgroundColor() {
            currentIndex = (currentIndex + 1) % colors.length;
            const chosen = colors[currentIndex];
            document.body.style.backgroundColor = chosen.hex;
            document.getElementById('currentColor').textContent = "Current Color: " + chosen.name + " (" + chosen.hex + ")";
        }

        setInterval(changeBackgroundColor, 5000);
    </script>
</body>
</html>

        const { useState } = React;

        function ColorSelector() {
            const [selectedColor, setSelectedColor] = useState('');

            return (
                <div className="app-card">
                    <h2>React Hooks Color Picker</h2>
                    <div className="btn-group">
                        <button className="btn-red" onClick={() => setSelectedColor('Red')}>Red</button>
                        <button className="btn-blue" onClick={() => setSelectedColor('Blue')}>Blue</button>

                            <span>Please click a color button above.</span>
                        )}

                </div>
            );
